% Noto Serif is optimized for long-form reading but does not cover every
% technical symbol used across the archive. Use a local system fallback for
% the few symbols observed in the corpus; no font files are bundled.
\newfontfamily\symbolfallback{DejaVu Sans}
\newunicodechar{≈}{{\symbolfallback ≈}}
\newunicodechar{⟨}{{\symbolfallback ⟨}}
\newunicodechar{⟩}{{\symbolfallback ⟩}}

\AtBeginDocument{\hypersetup{hidelinks}}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{%
  breaklines=true,%
  breakanywhere=true,%
  fontsize=\small%
}
\AtBeginDocument{%
  \ifcsname Highlighting\endcsname
    \RecustomVerbatimEnvironment{Highlighting}{Verbatim}{%
      commandchars=\\\{\},%
      breaklines=true,%
      breakanywhere=true,%
      fontsize=\small%
    }%
  \fi
}
